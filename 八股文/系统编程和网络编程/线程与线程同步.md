## 📚 一、线程创建与管理

### 📌 1. pthread_self () - 获取当前线程 ID

#### 头文件

```
#include <pthread.h>
```

#### 函数原型

```
pthread_t pthread_self(void);
```

#### 返回值

返回当前线程的 ID（pthread_t 类型）

#### 代码示例


```
#include <pthread.h>
#include <stdio.h>
#include <unistd.h>

void* thread_func(void* arg) {
    pthread_t tid = pthread_self();
    printf("[Thread] My thread ID: %lu\n", (unsigned long)tid);
    printf("[Thread] My process ID: %d\n", getpid());
    return NULL;
}

int main() {
    pthread_t tid;
    
    // 创建线程
    pthread_create(&tid, NULL, thread_func, NULL);
    
    // 主线程获取自己的ID
    pthread_t main_tid = pthread_self();
    printf("[Main] My thread ID: %lu\n", (unsigned long)main_tid);
    printf("[Main] My process ID: %d\n", getpid());
    
    // 等待子线程
    pthread_join(tid, NULL);
    
    return 0;
}
```

### 📌 2. pthread_create () - 创建线程

#### 头文件


```
#include <pthread.h>
```

#### 函数原型


```
int pthread_create(pthread_t *thread, const pthread_attr_t *attr,
                   void *(*start_routine)(void *), void *arg);
```

#### 参数说明

|参数|说明|
|---|---|
|thread|传出参数，保存新创建线程的 ID|
|attr|线程属性，NULL 表示使用默认属性|
|start_routine|线程执行函数（回调函数）|
|arg|传递给线程函数的参数|

#### 返回值

|返回值|说明|
|---|---|
|0|成功|
|非 0|失败，返回错误码（不是 errno）|

#### 代码示例 1：基本用法

```
#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void* thread_func(void* arg) {
    int num = *(int*)arg;
    printf("[Thread %d] Started, thread ID: %lu\n", num, (unsigned long)pthread_self());
    sleep(2);
    printf("[Thread %d] Finished\n", num);
    return (void*)(long)(num * 100);
}

int main() {
    pthread_t tid1, tid2;
    int arg1 = 1, arg2 = 2;
    void* retval;
    
    // 创建线程1
    int ret = pthread_create(&tid1, NULL, thread_func, &arg1);
    if (ret != 0) {
        fprintf(stderr, "pthread_create error: %s\n", strerror(ret));
        exit(1);
    }
    
    // 创建线程2
    ret = pthread_create(&tid2, NULL, thread_func, &arg2);
    if (ret != 0) {
        fprintf(stderr, "pthread_create error: %s\n", strerror(ret));
        exit(1);
    }
    
    // 等待线程1
    pthread_join(tid1, &retval);
    printf("[Main] Thread 1 returned: %ld\n", (long)retval);
    
    // 等待线程2
    pthread_join(tid2, &retval);
    printf("[Main] Thread 2 returned: %ld\n", (long)retval);
    
    printf("[Main] All threads finished\n");
    return 0;
}
```

#### 代码示例 2：传递多个参数

```
#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>

// 定义结构体传递多个参数
typedef struct {
    int id;
    char name[32];
    double value;
} ThreadArgs;

void* thread_func(void* arg) {
    ThreadArgs* args = (ThreadArgs*)arg;
    
    printf("[Thread %d] Name: %s, Value: %.2f\n", 
           args->id, args->name, args->value);
    
    // 修改参数（因为是共享内存）
    args->value *= 2;
    
    return NULL;
}

    pthread_t tid;
    ThreadArgs args = {1, "Worker", 3.14};
    
    pthread_create(&tid, NULL, thread_func, &args);
    pthread_join(tid, NULL);
    
    printf("[Main] Modified value: %.2f\n", args.value);
    
    return 0;
}
```

### 📌 3. pthread_exit () - 退出线程

#### 头文件


```
#include <pthread.h>
```

#### 函数原型


```
void pthread_exit(void *retval);
```

#### 参数说明

|参数|说明|
|---|---|
|retval|线程的返回值，可通过 pthread_join () 获取|

#### 返回值

无返回值（调用后线程终止）

#### 代码示例

```
#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>

void* thread_func(void* arg) {
    int* data = (int*)arg;
    
    if (*data < 0) {
        // 异常情况，直接退出
        printf("[Thread] Invalid data, exiting...\n");
        pthread_exit((void*)-1);
    }
    
    printf("[Thread] Processing data: %d\n", *data);
    *data *= 2;
    
    // 正常退出
    pthread_exit((void*)0);
}

int main() {
    pthread_t tid;
    int data = 10;
    void* retval;
    
    pthread_create(&tid, NULL, thread_func, &data);
    pthread_join(tid, &retval);
    
    if ((long)retval == 0) {
        printf("[Main] Thread succeeded, data = %d\n", data);
    } else {
        printf("[Main] Thread failed\n");
    }
    
    return 0;
}
```

### 📌 4. pthread_join () - 等待线程结束

#### 头文件

```
#include <pthread.h>
```

#### 函数原型

```
int pthread_join(pthread_t thread, void **retval);
```

#### 参数说明

|参数|说明|
|---|---|
|thread|要等待的线程 ID|
|retval|传出参数，接收线程的返回值（可为 NULL）|

#### 返回值

|返回值|说明|
|---|---|
|0|成功|
|非 0|失败，返回错误码|

#### 代码示例


```
#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

void* thread_func(void* arg) {
    int id = *(int*)arg;
    printf("[Thread %d] Started\n", id);
    
    sleep(id);
    
    printf("[Thread %d] Finished\n", id);
    return (void*)(long)(id * 10);
}

int main() {
    pthread_t tids[3];
    int ids[3] = {1, 2, 3};
    void* retval;
    
    // 创建3个线程
    for (int i = 0; i < 3; i++) {
        pthread_create(&tids[i], NULL, thread_func, &ids[i]);
    }
    
    // 按顺序等待线程结束
    for (int i = 0; i < 3; i++) {
        pthread_join(tids[i], &retval);
        printf("[Main] Thread %d returned: %ld\n", i + 1, (long)retval);
    }
    
    printf("[Main] All threads completed\n");
    return 0;
}
```

### 📌 5. pthread_detach () - 线程分离

#### 头文件

```
#include <pthread.h>
```

#### 函数原型

```
int pthread_detach(pthread_t thread);
```

#### 参数说明

|参数|说明|
|---|---|
|thread|要分离的线程 ID|

#### 返回值

|返回值|说明|
|---|---|
|0|成功|
|非 0|失败，返回错误码|

#### 代码示例

```
#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

void* thread_func(void* arg) {
    int id = *(int*)arg;
    printf("[Thread %d] Started (detached)\n", id);
    
    sleep(2);
    
    printf("[Thread %d] Finished\n", id);
    // 分离线程会自动释放资源，无需 pthread_join
    return NULL;
}

int main() {
    pthread_t tid;
    int id = 1;
    
    // 创建线程
    pthread_create(&tid, NULL, thread_func, &id);
    
    // 分离线程
    pthread_detach(tid);
    
    printf("[Main] Thread detached, continuing...\n");
    
    // 主线程继续执行
    sleep(3);
    
    printf("[Main] Main thread finished\n");
    return 0;
}
```

#### 头文件

```

#### 函数原型

```

```

#### 参数说明

|参数|说明|
|---|---|
|thread|要取消的线程 ID|

#### 返回值

|返回值|说明|
|---|---|
|0|成功|
|非 0|失败，返回错误码|

#### 代码示例

```
#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

void* thread_func(void* arg) {

    printf("[Thread %d] Started\n", id);
    
    for (int i = 0; i < 10; i++) {
        printf("[Thread %d] Working... %d\n", id, i);
        sleep(1);
        
        // 手动设置取消点
        pthread_testcancel();
    }
    
    printf("[Thread %d] Finished\n", id);
    return NULL;
}

int main() {
    pthread_t tid;
    int id = 1;
    void* retval;
    
    // 创建线程
    pthread_create(&tid, NULL, thread_func, &id);
    
    // 等待3秒后取消线程
    sleep(3);
    printf("[Main] Canceling thread...\n");
    
    pthread_cancel(tid);
    
    // 等待线程结束
    pthread_join(tid, &retval);
    
    if (retval == PTHREAD_CANCELED) {
        printf("[Main] Thread was canceled\n");
    }
    
    return 0;
}
```

### 📌 1. 互斥锁（Mutex）

#### pthread 互斥锁

##### 头文件：

```
#include <pthread.h>
```

##### 相关函数：

```
int pthread_mutex_init(pthread_mutex_t *mutex, const pthread_mutexattr_t *attr);
int pthread_mutex_destroy(pthread_mutex_t *mutex);
int pthread_mutex_lock(pthread_mutex_t *mutex);
int pthread_mutex_unlock(pthread_mutex_t *mutex);
int pthread_mutex_trylock(pthread_mutex_t *mutex);
```

##### 代码示例：

```
#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

#define NUM_THREADS 5
#define ITERATIONS 10000

int counter = 0;
pthread_mutex_t mutex;

void* increment_counter(void* arg) {
    int thread_id = *(int*)arg;
    
    for (int i = 0; i < ITERATIONS; i++) {
        pthread_mutex_lock(&mutex);
        counter++;
        pthread_mutex_unlock(&mutex);
    }
    
    printf("[Thread %d] Finished, counter = %d\n", thread_id, counter);
    return NULL;
}

int main() {
    pthread_t threads[NUM_THREADS];
    int thread_ids[NUM_THREADS];
    
    // 初始化互斥锁
    pthread_mutex_init(&mutex, NULL);
    
    // 创建线程
    for (int i = 0; i < NUM_THREADS; i++) {
        thread_ids[i] = i + 1;
        pthread_create(&threads[i], NULL, increment_counter, &thread_ids[i]);
    }
    
    // 等待所有线程
    for (int i = 0; i < NUM_THREADS; i++) {
        pthread_join(threads[i], NULL);
    }
    
    // 销毁互斥锁
    pthread_mutex_destroy(&mutex);
    
    printf("[Main] Final counter value: %d\n", counter);
    printf("[Main] Expected value: %d\n", NUM_THREADS * ITERATIONS);
    
    return 0;
}
```

#### C++ 互斥锁

##### 头文件：

#include <mutex>
#include <thread>
#include <iostream>
#include <vector>

std::mutex mtx;
int counter = 0;
const int NUM_THREADS = 5;
const int ITERATIONS = 10000;

void increment_counter(int thread_id) {
    for (int i = 0; i < ITERATIONS; i++) {
        std::lock_guard<std::mutex> lock(mtx);  // RAII，自动加锁解锁
        counter++;
    }
    
    std::cout << "[Thread " << thread_id << "] Finished" << std::endl;
}

int main() {
    std::vector<std::thread> threads;
    
    // 创建线程
    for (int i = 0; i < NUM_THREADS; i++) {
        threads.emplace_back(increment_counter, i + 1);
    }
    
    // 等待所有线程
    for (auto& t : threads) {
        t.join();
    }
    
    std::cout << "[Main] Final counter value: " << counter << std::endl;
    std::cout << "[Main] Expected value: " << NUM_THREADS * ITERATIONS << std::endl;
    
    return 0;
}

##### 代码示例：生产者 - 消费者模型

```
#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <stdbool.h>

#define BUFFER_SIZE 5

int buffer[BUFFER_SIZE];
int count = 0;
int in = 0, out = 0;

pthread_mutex_t mutex;
pthread_cond_t not_full;
pthread_cond_t not_empty;

void* producer(void* arg) {
    int id = *(int*)arg;
    
    for (int i = 0; i < 10; i++) {
        pthread_mutex_lock(&mutex);
        
        // 等待缓冲区不满
        while (count == BUFFER_SIZE) {
            pthread_cond_wait(&not_full, &mutex);
        }
        
        // 生产数据
        buffer[in] = i;
        in = (in + 1) % BUFFER_SIZE;
        count++;
        
        printf("[Producer %d] Produced: %d, count = %d\n", id, i, count);
        
        // 唤醒消费者
        pthread_cond_signal(&not_empty);
        
        pthread_mutex_unlock(&mutex);
        
        usleep(100000);  // 模拟生产时间
    }
    
    return NULL;
}

void* consumer(void* arg) {
    int id = *(int*)arg;
    
    for (int i = 0; i < 10; i++) {
        pthread_mutex_lock(&mutex);
        
        // 等待缓冲区不空
        while (count == 0) {
            pthread_cond_wait(&not_empty, &mutex);
        }
        
        // 消费数据
        int data = buffer[out];
        out = (out + 1) % BUFFER_SIZE;
        count--;
        
        printf("[Consumer %d] Consumed: %d, count = %d\n", id, data, count);
        
        // 唤醒生产者
        pthread_cond_signal(&not_full);
        
        pthread_mutex_unlock(&mutex);
        
        usleep(150000);  // 模拟消费时间
    }
    
    return NULL;
}

int main() {
    pthread_t producers[2], consumers[2];
    int prod_ids[2] = {1, 2};
    int cons_ids[2] = {1, 2};
    
    // 初始化
    pthread_mutex_init(&mutex, NULL);
    pthread_cond_init(&not_full, NULL);
    pthread_cond_init(&not_empty, NULL);
    
    // 创建生产者线程
    for (int i = 0; i < 2; i++) {
        pthread_create(&producers[i], NULL, producer, &prod_ids[i]);
    }
    
    // 创建消费者线程
    for (int i = 0; i < 2; i++) {
        pthread_create(&consumers[i], NULL, consumer, &cons_ids[i]);
    }
    
    // 等待所有线程
    for (int i = 0; i < 2; i++) {
        pthread_join(producers[i], NULL);
        pthread_join(consumers[i], NULL);
    }
    
    // 清理
    pthread_mutex_destroy(&mutex);
    pthread_cond_destroy(&not_full);
    pthread_cond_destroy(&not_empty);
    
    printf("[Main] All threads finished\n");
    return 0;
}
```

    for (int i = 0; i < 10; i++) {
        std::unique_lock<std::mutex> lock(mtx);
        
        // 等待缓冲区不满
        not_full.wait(lock, [] { return buffer.size() < BUFFER_SIZE; });
        
        // 生产数据
        buffer.push(i);
        std::cout << "[Producer " << id << "] Produced: " << i 
                  << ", size = " << buffer.size() << std::endl;
        
        // 唤醒消费者
        not_empty.notify_one();
        
        lock.unlock();
        std::this_thread::sleep_for(std::chrono::milliseconds(100));
    }
}

void consumer(int id) {
    for (int i = 0; i < 10; i++) {
        std::unique_lock<std::mutex> lock(mtx);
        
        // 等待缓冲区不空
        not_empty.wait(lock, [] { return !buffer.empty(); });
        
        // 消费数据
        int data = buffer.front();
        buffer.pop();
        std::cout << "[Consumer " << id << "] Consumed: " << data 
                  << ", size = " << buffer.size() << std::endl;
        
        // 唤醒生产者
        not_full.notify_one();
        
        lock.unlock();
        std::this_thread::sleep_for(std::chrono::milliseconds(150));
    }
}

int main() {
    std::vector<std::thread> threads;
    
    // 创建2个生产者和2个消费者
    for (int i = 0; i < 2; i++) {
        threads.emplace_back(producer, i + 1);
    }
    
    for (int i = 0; i < 2; i++) {
        threads.emplace_back(consumer, i + 1);
    }
    
    // 等待所有线程
    for (auto& t : threads) {
        t.join();
    }
    
    std::cout << "[Main] All threads finished" << std::endl;
    return 0;
}
```

##### 代码示例：线程池

```
#include <semaphore.h>
#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <stdbool.h>

#define MAX_TASKS 10
#define NUM_WORKERS 3

typedef struct {
    int id;
    bool valid;
} Task;

Task task_queue[MAX_TASKS];
int task_count = 0;
int task_in = 0, task_out = 0;

sem_t empty_sem;   // 空槽信号量
sem_t full_sem;    // 满槽信号量
sem_t mutex_sem;   // 互斥信号量

void* worker(void* arg) {
    int id = *(int*)arg;
    
    while (1) {
        // 等待任务
        sem_wait(&full_sem);
        sem_wait(&mutex_sem);
        
        // 取出任务
        Task task = task_queue[task_out];
        task_out = (task_out + 1) % MAX_TASKS;
        task_count--;
        
        sem_post(&mutex_sem);
        sem_post(&empty_sem);
        
        // 处理任务
        if (!task.valid) {
            printf("[Worker %d] Exiting\n", id);
            break;
        }
        
        printf("[Worker %d] Processing task %d\n", id, task.id);
        usleep(500000);  // 模拟处理时间
    }
    
    return NULL;
}

void add_task(int id) {
    sem_wait(&empty_sem);
    sem_wait(&mutex_sem);
    
    // 添加任务
    task_queue[task_in].id = id;
    task_queue[task_in].valid = true;
    task_in = (task_in + 1) % MAX_TASKS;
    task_count++;
    
    sem_post(&mutex_sem);
    sem_post(&full_sem);
}

int main() {
    pthread_t workers[NUM_WORKERS];
    int worker_ids[NUM_WORKERS];
    
    // 初始化信号量
    sem_init(&empty_sem, 0, MAX_TASKS);
    sem_init(&full_sem, 0, 0);
    sem_init(&mutex_sem, 0, 1);
    
    // 创建工作线程
    for (int i = 0; i < NUM_WORKERS; i++) {
        worker_ids[i] = i + 1;
        pthread_create(&workers[i], NULL, worker, &worker_ids[i]);
    }
    
    // 添加任务
    for (int i = 0; i < 15; i++) {
        printf("[Main] Adding task %d\n", i);
        add_task(i);
        usleep(200000);
    }
    
    // 添加结束任务
    for (int i = 0; i < NUM_WORKERS; i++) {
        sem_wait(&empty_sem);
        sem_wait(&mutex_sem);
        
        task_queue[task_in].valid = false;
        task_in = (task_in + 1) % MAX_TASKS;
        task_count++;
        
        sem_post(&mutex_sem);
        sem_post(&full_sem);
    }
    
    // 等待工作线程
    for (int i = 0; i < NUM_WORKERS; i++) {
        pthread_join(workers[i], NULL);
    }
    
    // 销毁信号量
    sem_destroy(&empty_sem);
    sem_destroy(&full_sem);
    sem_destroy(&mutex_sem);
    
    printf("[Main] All workers finished\n");
    return 0;
}
```

## 📚 五、线程安全与注意事项

### ⚠️ 常见问题与解决方案

#### 1. 竞态条件（Race Condition）

##### 问题：

```
// ❌ 非线程安全
int counter = 0;

void* increment(void* arg) {
    for (int i = 0; i < 10000; i++) {
        counter++;  // 非原子操作
    }
    return NULL;
}
```

##### 解决方案：

```
// ✅ 使用互斥锁
pthread_mutex_t mutex = PTHREAD_MUTEX_INITIALIZER;
int counter = 0;

void* increment(void* arg) {
    for (int i = 0; i < 10000; i++) {
        pthread_mutex_lock(&mutex);
        counter++;
        pthread_mutex_unlock(&mutex);
    }
    return NULL;
}
```

#### 2. 死锁（Deadlock）

##### 问题：

```
// ❌ 可能导致死锁
pthread_mutex_t mutex1 = PTHREAD_MUTEX_INITIALIZER;
pthread_mutex_t mutex2 = PTHREAD_MUTEX_INITIALIZER;

void* thread1_func(void* arg) {
    pthread_mutex_lock(&mutex1);
    pthread_mutex_lock(&mutex2);  // 如果 thread2 先锁了 mutex2，会死锁
    // ...
    pthread_mutex_unlock(&mutex2);
    pthread_mutex_unlock(&mutex1);
    return NULL;
}

void* thread2_func(void* arg) {
    pthread_mutex_lock(&mutex2);
    pthread_mutex_lock(&mutex1);  // 如果 thread1 先锁了 mutex1，会死锁
    // ...
    pthread_mutex_unlock(&mutex1);
    pthread_mutex_unlock(&mutex2);
    return NULL;
}
```

##### 解决方案：

```
// ✅ 按固定顺序加锁
void* thread1_func(void* arg) {
    pthread_mutex_lock(&mutex1);
    pthread_mutex_lock(&mutex2);
    // ...
    pthread_mutex_unlock(&mutex2);
    pthread_mutex_unlock(&mutex1);
    return NULL;
}

void* thread2_func(void* arg) {
    pthread_mutex_lock(&mutex1);  // 总是先锁 mutex1
    pthread_mutex_lock(&mutex2);
    // ...
    pthread_mutex_unlock(&mutex2);
    pthread_mutex_unlock(&mutex1);
    return NULL;
}
```

#### 3. 资源泄漏

##### 问题：

```
// ❌ 忘记 join 或 detach
void create_thread() {
    pthread_t tid;
    pthread_create(&tid, NULL, thread_func, NULL);
    // 忘记 pthread_join 或 pthread_detach
}
```

##### 解决方案：
```
// ✅ 确保资源回收
void create_thread() {
    pthread_t tid;
    pthread_create(&tid, NULL, thread_func, NULL);
    pthread_detach(tid);  // 或者稍后 pthread_join
}
```

## 📚 六、完整实战示例

### 💡 示例 1：线程池实现

```
#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <stdbool.h>

#define MAX_THREADS 5
#define MAX_QUEUE 10

typedef struct {
    void (*function)(void* arg);
    void* arg;
} Task;

typedef struct {
    Task tasks[MAX_QUEUE];
    int front, rear, count;
    pthread_mutex_t lock;
    pthread_cond_t not_empty;
    pthread_cond_t not_full;
    bool shutdown;
} ThreadPool;

ThreadPool pool;
pthread_t threads[MAX_THREADS];

void init_pool() {
    pool.front = 0;
    pool.rear = 0;
    pool.count = 0;
    pool.shutdown = false;
    pthread_mutex_init(&pool.lock, NULL);
    pthread_cond_init(&pool.not_empty, NULL);
    pthread_cond_init(&pool.not_full, NULL);
}

void add_task(void (*func)(void*), void* arg) {
    pthread_mutex_lock(&pool.lock);
    
    while (pool.count == MAX_QUEUE && !pool.shutdown) {
        pthread_cond_wait(&pool.not_full, &pool.lock);
    }
    
    if (pool.shutdown) {
        pthread_mutex_unlock(&pool.lock);
        return;
    }
    
    Task task = {func, arg};
    pool.tasks[pool.rear] = task;
    pool.rear = (pool.rear + 1) % MAX_QUEUE;
    pool.count++;
    
    pthread_cond_signal(&pool.not_empty);
    pthread_mutex_unlock(&pool.lock);
}

void* worker(void* arg) {
    int id = *(int*)arg;
    
    while (1) {
        pthread_mutex_lock(&pool.lock);
        
        while (pool.count == 0 && !pool.shutdown) {
            pthread_cond_wait(&pool.not_empty, &pool.lock);
        }
        
        if (pool.count == 0 && pool.shutdown) {
            pthread_mutex_unlock(&pool.lock);
            printf("[Worker %d] Exiting\n", id);
            pthread_exit(NULL);
        }
        
        Task task = pool.tasks[pool.front];
        pool.front = (pool.front + 1) % MAX_QUEUE;
        pool.count--;
        
        pthread_cond_signal(&pool.not_full);
        pthread_mutex_unlock(&pool.lock);
        
        // 执行任务
        printf("[Worker %d] Executing task\n", id);
        task.function(task.arg);
    }
}

void task_function(void* arg) {
    int* data = (int*)arg;
    printf("  Processing data: %d\n", *data);
    free(data);
    usleep(500000);
}

void shutdown_pool() {
    pthread_mutex_lock(&pool.lock);
    pool.shutdown = true;
    pthread_cond_broadcast(&pool.not_empty);
    pthread_cond_broadcast(&pool.not_full);
    pthread_mutex_unlock(&pool.lock);
    
    for (int i = 0; i < MAX_THREADS; i++) {
        pthread_join(threads[i], NULL);
    }
    
    pthread_mutex_destroy(&pool.lock);
    pthread_cond_destroy(&pool.not_empty);
    pthread_cond_destroy(&pool.not_full);
}

int main() {
    init_pool();
    
    // 创建工作线程
    int thread_ids[MAX_THREADS];
    for (int i = 0; i < MAX_THREADS; i++) {
        thread_ids[i] = i + 1;
        pthread_create(&threads[i], NULL, worker, &thread_ids[i]);
    }
    
    // 添加任务
    for (int i = 0; i < 20; i++) {
        int* data = malloc(sizeof(int));
        *data = i;
        add_task(task_function, data);
        usleep(100000);
    }
    
    // 关闭线程池
    shutdown_pool();
    
    printf("[Main] Thread pool shutdown complete\n");
    return 0;
}
```

## 📊 pthread 与 C++ thread 对比总结

表格

|特性|pthread (C)|std::thread (C++)|
|---|---|---|
|语言|C|C++11+|
|异常安全|❌|✅|
|资源管理|手动|RAII（自动）|
|语法简洁性|较繁琐|简洁|
|与 STL 集成|差|好|
|性能|相同|相同|
|跨平台|POSIX 系统|跨平台|
|学习曲线|陡峭|平缓|

## 🎯 最佳实践建议

1. 优先使用 C++ thread：如果使用 C++，优先使用 std::thread 和相关工具
2. 使用 RAII：使用 std::lock_guard、std::unique_lock 自动管理锁
3. 避免共享状态：尽量减少线程间共享数据
4. 使用线程池：避免频繁创建销毁线程
5. 正确处理错误：pthread 返回错误码，不是 errno
6. 避免死锁：按固定顺序加锁，使用超时机制
7. 资源回收：确保调用 pthread_join () 或设置分离状态
