# 🏛️ 1. `setsid()` - 创建新会话

## 📌 头文件

c

运行

```
#include <unistd.h>
```

## 📌 函数原型

c

运行

```
pid_t setsid(void);
```

## 📌 参数说明

无参数

## 📌 返回值

表格

|返回值|说明|
|---|---|
|> 0|成功：返回新会话的会话 ID（SID），等于调用进程的 PID|
|-1|失败：返回 -1，并设置 errno<br><br>EPERM：调用进程是进程组组长|

## 📌 关键特性

- 创建新会话：调用进程成为新会话的会话首进程（session leader）
- 创建新进程组：调用进程成为新进程组的组长进程
- 脱离控制终端：新会话没有控制终端（controlling terminal）
- 要求：调用进程不能是进程组组长（通常通过 `fork()` 后在子进程中调用）

## 💡 代码示例：验证 setsid () 的效果

c

运行

```
#include <unistd.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/types.h>

int main() {
    printf("Before setsid():\n");
    printf("  PID: %d, PGID: %d, SID: %d\n", 
           getpid(), getpgrp(), getsid(0));
    
    pid_t sid = setsid();
    
    if (sid == -1) {
        perror("setsid failed");
        exit(1);
    }
    
    printf("\nAfter setsid():\n");
    printf("  PID: %d, PGID: %d, SID: %d\n", 
           getpid(), getpgrp(), getsid(0));
    printf("  New session ID: %d\n", sid);
    
    return 0;
}
```

# 🛡️ 2. 守护进程创建 - 完整步骤详解

## 📌 守护进程创建的 6 个标准步骤

表格

|步骤|目的|系统调用|
|---|---|---|
|1. fork + 父进程退出|使子进程在后台运行，脱离终端控制|`fork()`|
|2. `setsid()`|创建新会话，脱离控制终端|`setsid()`|
|3. `chdir("/")`|改变工作目录，防止占用可卸载文件系统|`chdir()`|
|4. `umask(0)`|重设文件权限掩码，确保文件权限可控|`umask()`|
|5. 关闭 / 重定向文件描述符|关闭标准输入输出，避免占用终端|`close()`, `dup2()`|
|6. 业务逻辑|执行守护进程的实际工作|`while(1)`|

# 🎯 守护进程完整实现

## 💡 代码示例 1：标准守护进程创建函数

c

运行

```
#include <unistd.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <string.h>
#include <errno.h>

/**
 * 创建守护进程的标准函数
 * @param nochdir: 0 - 改变工作目录到 /, 非0 - 保持当前工作目录
 * @param noclose: 0 - 重定向标准输入输出到 /dev/null, 非0 - 保持原样
 * @return: 成功返回 0, 失败返回 -1
 */
int daemonize(int nochdir, int noclose) {
    pid_t pid;
    int fd;
    
    // 步骤 1: fork 子进程，父进程退出
    pid = fork();
    if (pid < 0) {
        perror("fork failed");
        return -1;
    }
    if (pid > 0) {
        // 父进程退出
        exit(0);
    }
    
    // 此时子进程在后台运行
    
    // 步骤 2: 创建新会话
    if (setsid() < 0) {
        perror("setsid failed");
        return -1;
    }
    
    // 步骤 3: 改变工作目录（可选）
    if (!nochdir) {
        if (chdir("/") < 0) {
            perror("chdir failed");
            return -1;
        }
    }
    
    // 步骤 4: 重设文件权限掩码
    umask(0);
    
    // 步骤 5: 关闭/重定向标准文件描述符（可选）
    if (!noclose) {
        // 关闭标准输入
        close(STDIN_FILENO);
        
        // 打开 /dev/null 作为新的标准输入
        fd = open("/dev/null", O_RDWR);
        if (fd < 0) {
            perror("open /dev/null failed");
            return -1;
        }
        // 复制到标准输出和标准错误
        if (dup2(fd, STDOUT_FILENO) < 0) {
            perror("dup2 stdout failed");
            close(fd);
            return -1;
        }
        if (dup2(fd, STDERR_FILENO) < 0) {
            perror("dup2 stderr failed");
            close(fd);
            return -1;
        }
        // 关闭原始描述符
        if (fd > STDERR_FILENO) {
            close(fd);
        }
    }
    
    return 0;
}

int main() {
    // 创建守护进程
    if (daemonize(0, 0) < 0) {
        fprintf(stderr, "Failed to daemonize\n");
        exit(1);
    }
    
    // 步骤 6: 守护进程业务逻辑
    while (1) {
        // 这里执行守护进程的实际工作
        // 例如：监听端口、处理请求、定时任务等
        
        // 示例：每 5 秒记录一次日志
        FILE *log = fopen("/tmp/daemon.log", "a");
        if (log) {
            time_t now = time(NULL);
            fprintf(log, "[%s] Daemon is running, PID: %d\n", 
                    ctime(&now), getpid());
            fclose(log);
        }
        
        sleep(5);
    }
    
    return 0;
}
```

## 💡 代码示例 2：带信号处理的守护进程

c

运行

```
#include <unistd.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <signal.h>
#include <time.h>
#include <string.h>
#include <errno.h>

volatile sig_atomic_t running = 1;

// 信号处理函数
void signal_handler(int signum) {
    if (signum == SIGTERM || signum == SIGINT) {
        running = 0;
    }
}

// 创建守护进程
void create_daemon() {
    pid_t pid;
    
    // 1. fork + 父进程退出
    pid = fork();
    if (pid < 0) {
        perror("fork failed");
        exit(1);
    }
    if (pid > 0) {
        exit(0);
    }
    
    // 2. 创建新会话
    if (setsid() < 0) {
        perror("setsid failed");
        exit(1);
    }
    
    // 3. 改变工作目录
    if (chdir("/") < 0) {
        perror("chdir failed");
        exit(1);
    }
    
    // 4. 重设 umask
    umask(022);
    
    // 5. 关闭/重定向文件描述符
    close(STDIN_FILENO);
    close(STDOUT_FILENO);
    close(STDERR_FILENO);
    
    int fd = open("/dev/null", O_RDWR);
    if (fd >= 0) {
        dup2(fd, STDIN_FILENO);
        dup2(fd, STDOUT_FILENO);
        dup2(fd, STDERR_FILENO);
        if (fd > 2) close(fd);
    }
}

// 写入日志
void write_log(const char *message) {
    FILE *log = fopen("/var/log/mydaemon.log", "a");
    if (log) {
        time_t now = time(NULL);
        char *time_str = ctime(&now);
        time_str[strlen(time_str) - 1] = '\0'; // 去除换行符
        
        fprintf(log, "[%s] %s (PID: %d)\n", time_str, message, getpid());
        fclose(log);
    }
}

int main() {
    // 创建守护进程
    create_daemon();
    
    // 设置信号处理
    struct sigaction sa;
    memset(&sa, 0, sizeof(sa));
    sa.sa_handler = signal_handler;
    sigemptyset(&sa.sa_mask);
    sa.sa_flags = 0;
    
    sigaction(SIGTERM, &sa, NULL);
    sigaction(SIGINT, &sa, NULL);
    
    // 记录守护进程启动
    write_log("Daemon started");
    
    // 6. 业务逻辑
    while (running) {
        // 模拟守护进程的工作
        // 例如：检查某个条件、处理请求等
        
        write_log("Daemon is running...");
        
        // 每 10 秒执行一次
        sleep(10);
    }
    
    // 清理工作
    write_log("Daemon stopped");
    
    return 0;
}
```

## 💡 代码示例 3：守护进程 + PID 文件

c

运行

```
#include <unistd.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <signal.h>
#include <string.h>
#include <errno.h>

#define PID_FILE "/var/run/mydaemon.pid"
#define LOG_FILE "/var/log/mydaemon.log"

volatile sig_atomic_t running = 1;

void signal_handler(int signum) {
    if (signum == SIGTERM || signum == SIGINT) {
        running = 0;
    }
}

void write_log(const char *message) {
    FILE *log = fopen(LOG_FILE, "a");
    if (log) {
        time_t now = time(NULL);
        char time_str[64];
        strftime(time_str, sizeof(time_str), "%Y-%m-%d %H:%M:%S", localtime(&now));
        
        fprintf(log, "[%s] %s (PID: %d)\n", time_str, message, getpid());
        fclose(log);
    }
}

// 创建 PID 文件
int create_pid_file() {
    int fd = open(PID_FILE, O_RDWR | O_CREAT, 0644);
    if (fd < 0) {
        write_log("Failed to create PID file");
        return -1;
    }
    
    // 尝试加锁，防止重复启动
    if (lockf(fd, F_TLOCK, 0) < 0) {
        if (errno == EACCES || errno == EAGAIN) {
            write_log("Daemon already running");
            close(fd);
            return -1;
        }
        close(fd);
        return -1;
    }
    
    // 写入 PID
    char pid_str[32];
    snprintf(pid_str, sizeof(pid_str), "%d\n", getpid());
    ftruncate(fd, 0);
    write(fd, pid_str, strlen(pid_str));
    
    return fd;
}

// 删除 PID 文件
void remove_pid_file(int fd) {
    if (fd >= 0) {
        close(fd);
        unlink(PID_FILE);
    }
}

void create_daemon() {
    pid_t pid = fork();
    if (pid < 0) exit(1);
    if (pid > 0) exit(0);
    
    if (setsid() < 0) exit(1);
    if (chdir("/") < 0) exit(1);
    umask(022);
    
    close(STDIN_FILENO);
    close(STDOUT_FILENO);
    close(STDERR_FILENO);
    
    int fd = open("/dev/null", O_RDWR);
    if (fd >= 0) {
        dup2(fd, STDIN_FILENO);
        dup2(fd, STDOUT_FILENO);
        dup2(fd, STDERR_FILENO);
        if (fd > 2) close(fd);
    }
}

int main() {
    // 创建守护进程
    create_daemon();
    
    // 创建 PID 文件
    int pid_fd = create_pid_file();
    if (pid_fd < 0) {
        exit(1);
    }
    
    // 设置信号处理
    struct sigaction sa;
    memset(&sa, 0, sizeof(sa));
    sa.sa_handler = signal_handler;
    sigemptyset(&sa.sa_mask);
    
    sigaction(SIGTERM, &sa, NULL);
    sigaction(SIGINT, &sa, NULL);
    
    write_log("Daemon started successfully");
    
    // 业务逻辑
    while (running) {
        write_log("Daemon is running...");
        sleep(5);
    }
    
    // 清理
    write_log("Daemon stopping...");
    remove_pid_file(pid_fd);
    
    return 0;
}
```

# 🎯 守护进程启动脚本示例

## 💡 System V 风格启动脚本 (/etc/init.d/mydaemon)

bash

运行

```
#!/bin/bash
# chkconfig: 2345 90 10
# description: My custom daemon

DAEMON="/usr/local/bin/mydaemon"
PIDFILE="/var/run/mydaemon.pid"
LOGFILE="/var/log/mydaemon.log"

case "$1" in
    start)
        echo "Starting mydaemon..."
        if [ -f $PIDFILE ]; then
            echo "Daemon already running"
            exit 1
        fi
        $DAEMON
        if [ $? -eq 0 ]; then
            echo "Daemon started"
        else
            echo "Failed to start daemon"
            exit 1
        fi
        ;;
    
    stop)
        echo "Stopping mydaemon..."
        if [ ! -f $PIDFILE ]; then
            echo "Daemon not running"
            exit 1
        fi
        kill -TERM $(cat $PIDFILE)
        rm -f $PIDFILE
        echo "Daemon stopped"
        ;;
    
    restart)
        $0 stop
        sleep 2
        $0 start
        ;;
    
    status)
        if [ -f $PIDFILE ]; then
            PID=$(cat $PIDFILE)
            if ps -p $PID > /dev/null; then
                echo "Daemon is running (PID: $PID)"
            else
                echo "Daemon not running, but PID file exists"
                rm -f $PIDFILE
            fi
        else
            echo "Daemon is not running"
        fi
        ;;
    
    *)
        echo "Usage: $0 {start|stop|restart|status}"
        exit 1
        ;;
esac

exit 0
```

# ⚠️ 守护进程最佳实践和注意事项

## ✅ 必须遵守的规则

1. 不要在守护进程中使用 `printf()`：标准输出已被重定向到 `/dev/null`
2. 使用日志文件记录信息：推荐使用 syslog 或自定义日志文件
3. 处理信号：至少处理 `SIGTERM` 和 `SIGINT` 以便优雅退出
4. 创建 PID 文件：防止重复启动，便于管理
5. 设置合理的 umask：确保创建的文件有正确的权限
6. 改变工作目录：避免占用可卸载的文件系统

## 🚫 常见错误

1. 忘记在子进程中调用 `setsid()`：导致无法脱离控制终端
2. 在进程组组长中调用 `setsid()`：会失败（EPERM 错误）
3. 没有关闭标准文件描述符：可能导致资源泄漏
4. 没有处理信号：无法优雅退出，只能强制 `kill -9`
5. 没有创建 PID 文件：无法判断进程是否已在运行

# 📝 完整的守护进程模板

c

运行

```
#include <unistd.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <signal.h>
#include <string.h>
#include <errno.h>
#include <time.h>

#define DAEMON_NAME "mydaemon"
#define PID_FILE "/var/run/" DAEMON_NAME ".pid"
#define LOG_FILE "/var/log/" DAEMON_NAME ".log"

volatile sig_atomic_t running = 1;

// 信号处理
void signal_handler(int signum) {
    switch (signum) {
        case SIGTERM:
        case SIGINT:
            running = 0;
            break;
        case SIGHUP:
            // 重新加载配置
            break;
    }
}

// 写日志
void log_message(const char *level, const char *format, ...) {
    FILE *log = fopen(LOG_FILE, "a");
    if (!log) return;
    
    time_t now = time(NULL);
    char time_str[64];
    strftime(time_str, sizeof(time_str), "%Y-%m-%d %H:%M:%S", localtime(&now));
    
    va_list args;
    va_start(args, format);
    fprintf(log, "[%s] [%s] ", time_str, level);
    vfprintf(log, format, args);
    fprintf(log, "\n");
    va_end(args);
    
    fclose(log);
}

// 创建守护进程
int daemonize() {
    pid_t pid = fork();
    if (pid < 0) return -1;
    if (pid > 0) exit(0); // 父进程退出
    
    if (setsid() < 0) return -1; // 创建新会话
    
    if (chdir("/") < 0) return -1; // 改变工作目录
    
    umask(022); // 设置 umask
    
    // 关闭标准文件描述符
    close(STDIN_FILENO);
    close(STDOUT_FILENO);
    close(STDERR_FILENO);
    
    // 重定向到 /dev/null
    int fd = open("/dev/null", O_RDWR);
    if (fd >= 0) {
        dup2(fd, STDIN_FILENO);
        dup2(fd, STDOUT_FILENO);
        dup2(fd, STDERR_FILENO);
        if (fd > 2) close(fd);
    }
    
    return 0;
}

// 创建 PID 文件
int create_pid_file() {
    int fd = open(PID_FILE, O_RDWR | O_CREAT, 0644);
    if (fd < 0) return -1;
    
    if (lockf(fd, F_TLOCK, 0) < 0) {
        close(fd);
        return -1;
    }
    
    char pid_str[32];
    snprintf(pid_str, sizeof(pid_str), "%d\n", getpid());
    ftruncate(fd, 0);
    write(fd, pid_str, strlen(pid_str));
    
    return fd;
}

// 删除 PID 文件
void remove_pid_file(int fd) {
    if (fd >= 0) {
        close(fd);
        unlink(PID_FILE);
    }
}

int main() {
    // 创建守护进程
    if (daemonize() < 0) {
        fprintf(stderr, "Failed to daemonize\n");
        exit(1);
    }
    
    // 创建 PID 文件
    int pid_fd = create_pid_file();
    if (pid_fd < 0) {
        log_message("ERROR", "Failed to create PID file, daemon may already be running");
        exit(1);
    }
    
    // 设置信号处理
    struct sigaction sa;
    memset(&sa, 0, sizeof(sa));
    sa.sa_handler = signal_handler;
    sigemptyset(&sa.sa_mask);
    
    sigaction(SIGTERM, &sa, NULL);
    sigaction(SIGINT, &sa, NULL);
    sigaction(SIGHUP, &sa, NULL);
    
    log_message("INFO", "Daemon started successfully (PID: %d)", getpid());
    
    // 主循环
    while (running) {
        // 执行守护进程的实际工作
        log_message("INFO", "Daemon is running...");
        
        // 模拟工作
        sleep(10);
    }
    
    // 清理
    log_message("INFO", "Daemon stopping...");
    remove_pid_file(pid_fd);
    
    return 0;
}
```

# 📚 总结

表格

|步骤|函数|目的|
|---|---|---|
|1|`fork()`|创建子进程，父进程退出|
|2|`setsid()`|创建新会话，脱离控制终端|
|3|`chdir("/")`|改变工作目录|
|4|`umask(0)`|重设文件权限掩码|
|5|`close()` + `dup2()`|关闭 / 重定向标准文件描述符|
|6|`while(1)`|执行业务逻辑|

守护进程是 Linux 系统中非常重要的概念，掌握其创建方法对系统编程至关重要！
