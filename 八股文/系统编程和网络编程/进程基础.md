下面我按照头文件、函数原型、参数说明、返回值、代码示例的格式，详细讲解 fork、exec、wait 这三个核心进程控制函数。

# 🧬 1. fork () - 创建子进程

## 📌 头文件

```
#include <unistd.h>
#include <sys/types.h>  // 用于 pid_t
```

## 📌 函数原型


```
pid_t fork(void);
```

## 📌 返回值

|返回值|说明|
|---|---|
|> 0|父进程：返回值为子进程的 PID|
|= 0|子进程：返回值为 0|
|-1|失败：返回 -1，并设置 errno|

## 📌 关键特性

- **调用一次，返回两次**
- 子进程是父进程的完整副本（代码段、数据段、堆栈等）
- 父子进程共享文件描述符、当前工作目录、环境变量
- 子进程拥有独立的进程 ID、独立的内存空间（**写时复制**）

## 💡 代码示例 1：基础用法

```
#include <unistd.h>
#include <sys/types.h>
#include <stdio.h>
#include <stdlib.h>

int main() {
    pid_t pid = fork();
    
    if (pid == -1) {
        perror("fork failed");
        exit(1);
    } else if (pid == 0) {
        // 子进程
        printf("[Child] PID: %d, Parent PID: %d\n", getpid(), getppid());
        exit(0);
    } else {
        // 父进程
        printf("[Parent] PID: %d, Child PID: %d\n", getpid(), pid);
        wait(NULL); // 等待子进程结束
    }
    
    return 0;
}
```

## 💡 代码示例 2：循环创建 n 个子进程

```
#include <unistd.h>
#include <sys/types.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/wait.h>

int main() {
    int n = 5;
    
    for (int i = 0; i < n; i++) {
        pid_t pid = fork();
        
        if (pid == -1) {
            perror("fork failed");
            exit(1);
        } else if (pid == 0) {
            // 子进程
            printf("[Child %d] PID: %d, I'm child number %d\n", i + 1, getpid(), i + 1);
            sleep(i + 1); // 不同的睡眠时间
            exit(i + 1);
        }
        // 父进程继续循环创建下一个子进程
    }
    
    // 父进程等待所有子进程
    printf("[Parent] Waiting for all children...\n");
    for (int i = 0; i < n; i++) {
        pid_t pid = wait(NULL);
        printf("[Parent] Child %d terminated\n", pid);
    }
    
    return 0;
}
```

## 💡 代码示例 3：验证写时复制（Copy-on-Write）

```
#include <unistd.h>
#include <sys/types.h>
#include <stdio.h>
#include <stdlib.h>

int global_var = 100;

int main() {
    int local_var = 200;
    
    pid_t pid = fork();
    
    if (pid == 0) {
        // 子进程
        printf("[Child] Before: global_var=%d, local_var=%d, addr_global=%p, addr_local=%p\n", 
               global_var, local_var, &global_var, &local_var);
        
        global_var = 300;
        local_var = 400;
        
        printf("[Child] After:  global_var=%d, local_var=%d, addr_global=%p, addr_local=%p\n", 
               global_var, local_var, &global_var, &local_var);
        exit(0);
    } else {
        // 父进程
        sleep(1); // 等待子进程先执行
        
        printf("[Parent] global_var=%d, local_var=%d, addr_global=%p, addr_local=%p\n", 
               global_var, local_var, &global_var, &local_var);
               
        global_var = 500;
        local_var = 600;
        
        printf("[Parent] global_var=%d, local_var=%d, addr_global=%p, addr_local=%p\n", 
               global_var, local_var, &global_var, &local_var);
        
        wait(NULL);
    }
    
    return 0;
}
```

>**写时复制**：
>fork 后父子共享虚拟地址、共享物理内存；
>只有写数据时，才给修改方**分配新物理内存**，**虚拟地址始终不变**(只会**修改页表项映射**)。
>
>你打印的 `%p` 输出的是 **虚拟地址**，不是物理地址！
>
>结果：打印的四次虚拟地址完全一样！


# 🚀 2. exec 函数族 - 执行新程序

## 📌 头文件


```
#include <unistd.h>
```

## 📌 函数原型（6 个变体）

| 函数         | 原型                                                                       | 说明                 |
| ---------- | ------------------------------------------------------------------------ | ------------------ |
| **execl**  | `int execl(const char *path, const char *arg, ...)`                      | 路径 + 逐个参数          |
| **execlp** | `int execlp(const char *file, const char *arg, ...)`                     | 文件名 + 逐个参数（查 PATH） |
| execle     | `int execle(const char *path, const char *arg, ..., char *const envp[])` | 路径 + 逐个参数 + 环境变量   |
| execv      | `int execv(const char *path, char *const argv[])`                        | 路径 + 参数数组          |
| execvp     | `int execvp(const char *file, char *const argv[])`                       | 文件名 + 参数数组（查 PATH） |
| execvpe    | `int execvpe(const char *file, char *const argv[], char *const envp[])`  | 文件名 + 参数数组 + 环境变量  |

## 📌 命名规则

- `l`（list）：参数逐个列出，以 NULL 结尾
- `v`（vector）：参数以数组形式传入
- `p`（path）：自动在 PATH 环境变量中查找程序
- `e`（env）：可以指定新的环境变量

## 📌 参数说明

|参数|说明|
|---|---|
|path|程序的绝对路径或相对路径|
|file|程序名（exec*p 会在 PATH 中查找）|
|arg / argv|程序的参数列表，argv [0] 通常是程序名|
|envp|环境变量数组（exec*e 系列）|

## 📌 返回值

| 返回值 | 说明                   |
| --- | -------------------- |
| 无返回 | **成功：进程被新程序替换，不会返回** |
| -1  | 失败：返回 -1，并设置 errno   |

## 📌 关键特性

- 替换当前进程：成功后，当前进程的代码段、数据段、堆栈被新程序完全替换->后续的代码不会继续执行
- 保留的属性：进程 ID、父进程 ID、打开的文件描述符（除非设置了 FD_CLOEXEC）、当前工作目录
- 重置的属性：代码段、数据段、堆栈、信号处理函数（恢复默认）

## 💡 代码示例 1：execlp - 执行系统命令

```
#include <unistd.h>
#include <stdio.h>
#include <stdlib.h>

int main() {
    printf("Before exec...\n");
    
    // 执行 ls -l /tmp 命令
    execlp("ls", "ls", "-l", "/tmp", NULL);
    
    // 如果 exec 成功，下面的代码不会执行
    perror("execlp failed"); // 只有失败才到这里
    exit(1);
}
```

## 💡 代码示例 2：execv - 执行自定义程序

```
#include <unistd.h>
#include <stdio.h>
#include <stdlib.h>

int main() {
    char *args[] = {
        "./myprogram",  // argv[0] - 程序名
        "arg1",         // argv[1]
        "arg2",         // argv[2]
        NULL            // 必须以 NULL 结尾
    };
    
    // 执行当前目录下的 myprogram
    execv("./myprogram", args);
    
    perror("execv failed");
    exit(1);
}
```

## 💡 代码示例 5：完整流程 - fork + exec

```
#include <unistd.h>
#include <sys/types.h>
#include <sys/wait.h>
#include <stdio.h>
#include <stdlib.h>

int main() {
    pid_t pid = fork();
    
    if (pid == -1) {
        perror("fork failed");
        exit(1);
    } else if (pid == 0) {
        // 子进程：执行新程序
        printf("[Child] Executing ls -l\n");
        
        char *args[] = {"ls", "-l", NULL};
        execvp(args[0], args);
        
        // 如果到这里，说明 exec 失败
        perror("execvp failed");
        exit(1);
    } else {
        // 父进程：等待子进程
        printf("[Parent] Waiting for child...\n");
        wait(NULL);
        printf("[Parent] Child finished\n");
    }
    
    return 0;
}
```

# 🔄 3. wait () - 等待子进程结束

## 📌 头文件

```
#include <sys/wait.h>
#include <sys/types.h>
```

## 📌 函数原型

```
pid_t wait(int *status);
```

## 📌 参数说明

| 参数     | 说明                                                                         |
| ------ | -------------------------------------------------------------------------- |
| status | 传出参数：用于获取子进程的退出状态<br><br>NULL：不关心子进程退出状态<br><br>非 NULL：指向一个 int 变量，存储子进程状态 |

## 📌 返回值

| 返回值 | 说明                                                           |
| --- | ------------------------------------------------------------ |
| > 0 | 成功：返回**被回收的子进程 PID**                                         |
| -1  | 失败：返回 -1，并设置 errno<br><br>ECHILD：没有子进程可回收<br><br>EINTR：被信号中断 |

## 📌 关键特性

- 阻塞等待：父进程会**阻塞**，直到任意一个子进程结束
- 回收资源：清理子进程的 PCB（进程控制块），**防止僵尸进程**
- 获取状态：通过 status 参数获取子进程退出信息

## 📌 状态检查宏

```
WIFEXITED(status)    // 子进程是否正常退出
WEXITSTATUS(status)  // 获取正常退出的返回值（0-255）

WIFSIGNALED(status)  // 子进程是否被信号终止
WTERMSIG(status)     // 获取终止子进程的信号编号

WIFSTOPPED(status)   // 子进程是否被信号暂停
WSTOPSIG(status)     // 获取暂停子进程的信号编号

WIFCONTINUED(status) // 子进程是否被信号继续
```

## 💡 代码示例 1：基础用法

```
#include <sys/wait.h>
#include <unistd.h>
#include <sys/types.h>
#include <stdio.h>
#include <stdlib.h>

int main() {
    pid_t pid = fork();
    
    if (pid == 0) {
        // 子进程
        printf("[Child] PID: %d\n", getpid());
        sleep(2);
        exit(42); // 返回值 42
    } else {
        // 父进程
        int status;
        pid_t child_pid = wait(&status);
        
        printf("[Parent] Child %d terminated\n", child_pid);
        
        if (WIFEXITED(status)) {
            printf("[Parent] Child exited normally with code: %d\n", 
                   WEXITSTATUS(status));
        } else if (WIFSIGNALED(status)) {
            printf("[Parent] Child terminated by signal: %d\n", 
                   WTERMSIG(status));
        }
    }
    
    return 0;
}
```

## 💡 代码示例 2：回收多个子进程

```
#include <sys/wait.h>
#include <unistd.h>
#include <sys/types.h>
#include <stdio.h>
#include <stdlib.h>

int main() {
    int n = 3;
    
    // 创建 3 个子进程
    for (int i = 0; i < n; i++) {
        pid_t pid = fork();
        
        if (pid == 0) {
            printf("[Child %d] PID: %d\n", i + 1, getpid());
            sleep(i + 1);
            exit(i + 1);
        }
    }
    
    // 父进程回收所有子进程
    int status;
    pid_t pid;
    
    printf("[Parent] Waiting for children...\n");
    
    while ((pid = wait(&status)) > 0) {
        if (WIFEXITED(status)) {
            printf("[Parent] Child %d exited with code %d\n", 
                   pid, WEXITSTATUS(status));
        }
    }
    
    printf("[Parent] All children terminated\n");
    
    return 0;
}
```

## 💡 代码示例 3：处理被信号终止的子进程

```
#include <sys/wait.h>
#include <unistd.h>
#include <sys/types.h>
#include <stdio.h>
#include <stdlib.h>
#include <signal.h>

int main() {
    pid_t pid = fork();
    
    if (pid == 0) {
        // 子进程
        printf("[Child] PID: %d, sleeping...\n", getpid());
        sleep(10); // 长时间睡眠
        exit(0);
    } else {
        // 父进程：发送信号终止子进程
        sleep(1);
        printf("[Parent] Sending SIGTERM to child %d\n", pid);
        kill(pid, SIGTERM);
        
        int status;
        wait(&status);
        
        if (WIFSIGNALED(status)) {
            printf("[Parent] Child terminated by signal %d\n", WTERMSIG(status));
            
            #ifdef WCOREDUMP
            if (WCOREDUMP(status)) {
                printf("[Parent] Core dump generated\n");
            }
            #endif
        }
    }
    
    return 0;
}
```

# 🎯 4. waitpid () - 等待指定子进程

## 📌 头文件

```
#include <sys/wait.h>
#include <sys/types.h>
```

## 📌 函数原型

```
pid_t waitpid(pid_t pid, int *status, int options);
```

## 📌 参数说明

|参数|说明|
|---|---|
|pid|指定等待的子进程：<br><br>> 0：等待指定 PID 的子进程<br><br>= 0：等待同一进程组的任一子进程<br><br>= -1：等待任意子进程（等同于 wait ()）<br><br>< -1：等待指定进程组的任一子进程|
|status|传出参数：子进程退出状态（同 wait ()）|
|options|等待选项：<br><br>0：阻塞等待<br><br>WNOHANG：非阻塞，若无子进程退出立即返回<br><br>WUNTRACED：报告暂停的子进程<br><br>WCONTINUED：报告继续的子进程|

## 📌 返回值

| 返回值 | 说明                         |
| --- | -------------------------- |
| > 0 | 成功：返回**被回收的子进程 PID**       |
| = 0 | 使用 **WNOHANG** 且**无子进程退出** |
| -1  | 失败：返回 -1，并设置 errno         |

## 💡 代码示例 1：非阻塞等待（WNOHANG）

```
#include <sys/wait.h>
#include <unistd.h>
#include <sys/types.h>
#include <stdio.h>
#include <stdlib.h>

int main() {
    pid_t pid = fork();
    
    if (pid == 0) {
        // 子进程
        printf("[Child] PID: %d, sleeping for 3 seconds...\n", getpid());
        sleep(3);
        exit(100);
    } else {
        // 父进程：非阻塞等待
        int status;
        pid_t result;
        
        printf("[Parent] Checking child status...\n");
        
        while (1) {
            result = waitpid(pid, &status, WNOHANG);
            
            if (result == 0) {
                // 子进程还在运行
                printf("[Parent] Child still running, doing other work...\n");
                sleep(1);
            } else if (result > 0) {
                // 子进程已退出
                printf("[Parent] Child %d terminated\n", result);
                
                if (WIFEXITED(status)) {
                    printf("[Parent] Exit code: %d\n", WEXITSTATUS(status));
                }
                break;
            } else {
                // 错误
                perror("waitpid failed");
                break;
            }
        }
    }
    
    return 0;
}
```

## 💡 代码示例 2：等待指定子进程

```
#include <sys/wait.h>
#include <unistd.h>
#include <sys/types.h>
#include <stdio.h>
#include <stdlib.h>

int main() {
    pid_t pid1 = fork();
    
    if (pid1 == 0) {
        printf("[Child 1] PID: %d\n", getpid());
        sleep(2);
        exit(10);
    }
    
    pid_t pid2 = fork();
    
    if (pid2 == 0) {
        printf("[Child 2] PID: %d\n", getpid());
        sleep(1);
        exit(20);
    }
    
    // 父进程：先等待 child 2，再等待 child 1
    int status;
    
    printf("[Parent] Waiting for child 2 (PID: %d)...\n", pid2);
    waitpid(pid2, &status, 0);
    printf("[Parent] Child 2 exited with code: %d\n", WEXITSTATUS(status));
    
    printf("[Parent] Waiting for child 1 (PID: %d)...\n", pid1);
    waitpid(pid1, &status, 0);
    printf("[Parent] Child 1 exited with code: %d\n", WEXITSTATUS(status));
    
    return 0;
}
```

# 📊 函数对比总结

| 函数        | 作用      | 阻塞性        | 返回值特点                |
| --------- | ------- | ---------- | -------------------- |
| fork()    | 创建子进程   | 不阻塞        | 父进程返回子进程 PID，子进程返回 0 |
| exec*()   | 替换当前进程  | 无返回（成功）    | 失败返回 -1              |
| wait()    | 等待任意子进程 | 阻塞         | 返回被回收的子进程 PID        |
| waitpid() | 等待指定子进程 | 可选阻塞 / 非阻塞 | 返回被回收的子进程 PID        |

# 🎓 综合实战：实现简易 Shell

```
#include <unistd.h>
#include <sys/types.h>
#include <sys/wait.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_ARGS 10
#define BUFFER_SIZE 1024

int main() {
    char buffer[BUFFER_SIZE];
    char *args[MAX_ARGS];
    
    printf("Simple Shell (type 'exit' to quit)\n");
    
    while (1) {
        // 显示提示符
        printf("$ ");
        fflush(stdout);
        
        // 读取用户输入
        if (fgets(buffer, BUFFER_SIZE, stdin) == NULL) {
            break;
        }
        
        // 去除换行符
        buffer[strcspn(buffer, "\n")] = '\0';
        
        // 检查退出命令
        if (strcmp(buffer, "exit") == 0) {
            break;
        }
        
        // 解析命令参数
        int argc = 0;
        char *token = strtok(buffer, " ");
        
        while (token != NULL && argc < MAX_ARGS - 1) {
            args[argc++] = token;
            token = strtok(NULL, " ");
        }
        args[argc] = NULL;
        
        if (argc == 0) {
            continue; // 空命令
        }
        
        // 创建子进程执行命令
        pid_t pid = fork();
        
        if (pid == -1) {
            perror("fork failed");
            continue;
        } else if (pid == 0) {
            // 子进程：执行命令
            execvp(args[0], args);
            
            // 如果到这里，说明 exec 失败
            fprintf(stderr, "Command not found: %s\n", args[0]);
            exit(1);
        } else {
            // 父进程：等待子进程
            wait(NULL);
        }
    }
    
    printf("Shell terminated\n");
    return 0;
}
```

# ⚠️ 重要注意事项

1. `fork()` 后的文件描述符：父子进程共享文件描述符，对文件偏移量的操作互相影响
2. `exec()` 失败处理：必须检查返回值，失败时通常需要 `exit()` 退出子进程
3. 僵尸进程：父进程必须调用 `wait()`/`waitpid()` 回收子进程，否则会产生僵尸进程
4. 信号处理：`wait()` 可能被信号中断（返回 -1，errno = EINTR）
5. 多子进程回收：需要循环调用 `wait()`/`waitpid()` 直到返回 -1 且 errno = ECHILD
