### **题目：146. LRU 缓存**

>请你设计并实现一个满足  [LRU (最近最少使用) 缓存](https://baike.baidu.com/item/LRU) 约束的数据结构。
>
>实现 `LRUCache` 类：
>- `LRUCache(int capacity)` 以 **正整数** 作为容量 `capacity` 初始化 LRU 缓存
>- `int get(int key)` 如果关键字 `key` 存在于缓存中，则返回关键字的值，否则返回 `-1` 。
>- `void put(int key, int value)` 如果关键字 `key` 已经存在，则变更其数据值 `value` ；如果不存在，则向缓存中插入该组 `key-value` 。如果插入操作导致关键字数量超过 `capacity` ，则应该 **逐出** 最久未使用的关键字。
>
>函数 `get` 和 `put` 必须以 `O(1)` 的平均时间复杂度运行。


#### 🔹 思路1

/*
1.为什么是双向循环链表？
提取节点(removeNode)时需要知道prev

2.为什么节点需要保存key？
从哈希表移除末尾node时需要key

3.为什么需要dummy？需要多少个？
简化代码逻辑 不需要判断初始无节点的情况 | 循环链表->一个即可  dummy->next为第一个节点 dummy->prev为最后一个节点
*/

//双向链表
struct Node
{
    int key;
    int val;
    Node* prev;
    Node* next;

    Node():key(-1),val(0),prev(nullptr),next(nullptr){}
    Node(int key,int val):key(key),val(val),prev(nullptr),next(nullptr){}
    Node(int key,int val,Node* prev,Node* next):key(key),val(val),prev(prev),next(next){}
};
class LRUCache
{
public:
    LRUCache(int capacity):capacity(capacity)
    {
        dummy=new Node;
        dummy->prev=dummy->next=dummy;
    }
    ~LRUCache()
    {
        delete dummy;
        for(auto &[_,node] : key_to_Node) delete node;
    }

    int get(int key)
    {
        Node* node=getNode(key);
        return node?node->val:-1;
    }   

    void put(int key,int val)
    {
        Node* node=getNode(key);
        //找到->更新位置+更新值
        if(node) node->val=val;
        //未找到->插入头部+插入哈希表+考虑容量(->末尾node移出链表和哈希表+释放空间)
        else
        {
            node=new Node(key,val);
            pushFrontNode(node);
            key_to_Node[key]=node;
            
            //考虑容量
            if(key_to_Node.size()>capacity)
            {
                Node* r=dummy->prev;
                key_to_Node.erase(r->key);
                removeNode(r);
                delete r;
            }
        }
    } 

private:
    //从链表中提取出node
    void removeNode(Node* node)
    {     
        node->prev->next=node->next;
        node->next->prev=node->prev;
    }
    //将node插入链表头部
    void pushFrontNode(Node* node)
    {
        node->prev=dummy;
        node->next=dummy->next;
        node->next->prev=node;
        dummy->next=node;
    }
    //根据key获取node
    Node* getNode(int key)
    {
        auto it=key_to_Node.find(key);
        if(it==key_to_Node.end()) return nullptr;
        
        Node* node=it->second;
        removeNode(node);       //提取
        pushFrontNode(node);    //插入头部
        return node;
    }
    int capacity;
    unordered_map<int,Node*>key_to_Node;
    Node* dummy;
};
